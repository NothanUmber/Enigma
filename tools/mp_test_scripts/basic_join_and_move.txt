# Basic direct-connect smoke/regression script:
# - force the direct path
# - start host session and join client
# - verify direct transport selection
# - load a level, start gameplay, and verify both local players can move

both SET_RELAY_SERVER server=-
both SET_TCP_RELAY_SERVER server=-
both SET_BOOL key=MultiplayerEnableDirect value=1
both SET_BOOL key=MultiplayerEnableUdpRelay value=0
both SET_BOOL key=MultiplayerEnableTcpRelay value=0

host START_HOST session=1 seed=1 expected=2 port=23456 host_id=mptest-host
client START_CLIENT session=1 seed=1 expected=2 port=23456 host_ip=127.0.0.1 host_id=mptest-host

wait role=client contains="EVT name=JOINED" timeout_ms=15000

both STREAM_STATE interval_ms=100
wait_state_value role=host field=mp_direct equals=1 timeout_ms=5000
wait_state_value role=host field=mp_udp equals=0 timeout_ms=5000
wait_state_value role=host field=mp_tcp equals=0 timeout_ms=5000
wait_state_value role=client field=mp_transport equals=1 timeout_ms=5000

# Pick a level by its `Proxy::getNormLevelPath()` (typically "<pack>/<base_name>").
host LOAD_LEVEL level_id="enigma_vii/ais05_1"
sleep 500
both START_GAME
wait_state_value role=host field=mp_phase equals=3 timeout_ms=5000
wait_state_value role=client field=mp_phase equals=3 timeout_ms=5000

host HOLD_MOUSE_FORCE player=0 ticks=20 fx=1.0 fy=0.0
client HOLD_MOUSE_FORCE player=1 ticks=20 fx=-1.0 fy=0.0
wait_state_change role=host field=p0_x min_abs_delta=0.005 timeout_ms=5000
wait_state_change role=client field=p1_x min_abs_delta=0.005 timeout_ms=5000

both QUIT
