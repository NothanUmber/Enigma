# Probe PuzzleStone world-resync parity for hidden SINGLE and ignite-alarm repair.

both SET_BOOL key=MultiplayerAutoDetectConnectivity value=0
both SET_BOOL key=MultiplayerDebugVisualPrediction value=0
both SET_BOOL key=MultiplayerDebugZeroFillInputs value=1
both SET_BOOL key=MultiplayerDebugRollbackEnabled value=0
both SET_BOOL key=MultiplayerDebugNetSimEnabled value=0
both SET_BOOL key=MultiplayerDebugLogging value=1
both SET_INT key=MultiplayerDebugTickLengthMs value=10
both SET_INT key=MultiplayerDebugInputDelayTicks value=0
both SET_INT key=MultiplayerDebugHostBroadcastResyncStrideTicks value=0
both SET_INT key=MultiplayerDebugHostBroadcastWorldStateStrideTicks value=1

host START_HOST session=1 seed=1 expected=2 port=23484 host_id=mptest-host
client START_CLIENT session=1 seed=1 expected=2 port=23484 host_ip=127.0.0.1 host_id=mptest-host
wait role=client contains="EVT name=JOINED" timeout_ms=15000

host LOAD_LEVEL level_id="enigma_experimental/mptest_puzzlestone_world_resync_1"
sleep 1500
both START_GAME
sleep 1000

# Hidden SINGLE-flag repair on isolated yellow puzzle stone.
both GET_PUZZLESTONE_STATE x=6 y=2
client FORCE_PUZZLESTONE_STATE x=6 y=2 single=1
client GET_PUZZLESTONE_STATE x=6 y=2
sleep 40
both GET_PUZZLESTONE_STATE x=6 y=2

# Hidden ignite-alarm repair on the right stone of a complete yellow pair.
host SEND_CELL_MESSAGE x=3 y=2 layer=st message=_cannonball
sleep 20
both GET_PUZZLESTONE_STATE x=4 y=2
client FORCE_PUZZLESTONE_STATE x=4 y=2 internal=4 clear_alarm=1
client GET_PUZZLESTONE_STATE x=4 y=2
sleep 40
both GET_PUZZLESTONE_STATE x=4 y=2
sleep 220
both GET_PUZZLESTONE_STATE x=4 y=2

both QUIT
