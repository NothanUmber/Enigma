# Regression for the Open Sesame door-coupled Per.Oxyd setup.
# The predicted client must diverge from current truth immediately after the
# local shogun push, persist that locally changed state, and later converge.

both SET_BOOL key=MultiplayerAutoDetectConnectivity value=0
both SET_BOOL key=MultiplayerDebugZeroFillInputs value=0
both SET_BOOL key=MultiplayerDebugVisualPrediction value=1
both SET_INT key=MultiplayerDebugPredictMissingMouseTicks value=0
both SET_INT key=MultiplayerDebugInputDelayTicks value=120
both SET_INT key=MultiplayerDebugTickLengthMs value=10
both SET_INT key=MultiplayerDebugNetSimDelayMs value=0
both SET_BOOL key=MultiplayerDebugRollbackEnabled value=0
both SET_BOOL key=MultiplayerDebugNetSimEnabled value=0

host START_HOST session=1 seed=1 expected=2 port=23456 host_id=mptest-host
client START_CLIENT session=1 seed=1 expected=2 port=23456 host_ip=127.0.0.1 host_id=mptest-host
wait role=client contains="EVT name=JOINED" timeout_ms=15000

host LOAD_LEVEL level_id="enigma_peroxyd/peroxyd04_2"
sleep 1500
both START_GAME
sleep 1000

both SET_ACTOR_POS player=1 x=4.50 y=4.50 vx=0.0 vy=0.0

# Put a large shogun just below the right-hand large dot so player 0 can push it up.
both SET_STONE x=10 y=7 kind=st_shogun_l

both SET_ACTOR_POS player=0 x=10.50 y=8.60 vx=0.0 vy=0.0
host HOLD_MOUSE_FORCE player=0 ticks=6 fx=0.0 fy=-220.0

sleep 80

host GET_CELL_RENDER x=10 y=6
wait_next role=host contains="OK cmd=GET_CELL_RENDER x=10 y=6 fl=fl_rough st=st_shogun_l it=it_shogun_l" timeout_ms=1500
client GET_CELL_RENDER x=10 y=6
wait_next role=client contains="OK cmd=GET_CELL_RENDER x=10 y=6 fl=fl_rough st=- it=it_shogun_l" timeout_ms=1500

host GET_CELL_RENDER x=10 y=7
wait_next role=host contains="OK cmd=GET_CELL_RENDER x=10 y=7 fl=fl_rough st=- it=-" timeout_ms=1500
client GET_CELL_RENDER x=10 y=7
wait_next role=client contains="OK cmd=GET_CELL_RENDER x=10 y=7 fl=fl_rough st=st_shogun_l it=-" timeout_ms=1500

sleep 500

host GET_CELL_RENDER x=10 y=6
wait_next role=host contains="OK cmd=GET_CELL_RENDER x=10 y=6 fl=fl_rough st=st_shogun_l it=it_shogun_l" timeout_ms=1500
client GET_CELL_RENDER x=10 y=6
wait_next role=client contains="OK cmd=GET_CELL_RENDER x=10 y=6 fl=fl_rough st=- it=it_shogun_l" timeout_ms=1500

sleep 1000

host GET_CELL_RENDER x=10 y=6
wait_next role=host contains="OK cmd=GET_CELL_RENDER x=10 y=6 fl=fl_rough st=st_shogun_l it=it_shogun_l" timeout_ms=1500
client GET_CELL_RENDER x=10 y=6
wait_next role=client contains="OK cmd=GET_CELL_RENDER x=10 y=6 fl=fl_rough st=st_shogun_l it=it_shogun_l" timeout_ms=1500

host GET_CELL_RENDER x=10 y=7
wait_next role=host contains="OK cmd=GET_CELL_RENDER x=10 y=7 fl=fl_rough st=- it=-" timeout_ms=1500
client GET_CELL_RENDER x=10 y=7
wait_next role=client contains="OK cmd=GET_CELL_RENDER x=10 y=7 fl=fl_rough st=- it=-" timeout_ms=1500
