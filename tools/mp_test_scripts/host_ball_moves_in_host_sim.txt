# Confirms the host-controlled ball actually moves on the host instance.
#
# This is intended to catch the regression where balls only move on the client
# simulation and then get snapped back to the host state.

host START_HOST session=1 seed=1 expected=2 port=23456 host_id=mptest-host
client START_CLIENT session=1 seed=1 expected=2 port=23456 host_ip=127.0.0.1 host_id=mptest-host
wait role=client contains="EVT name=JOINED" timeout_ms=15000

# A stable, simple built-in level with free space near the start.
host LOAD_LEVEL level_id="enigma_demolevels/ralD006_1"
sleep 800
both START_GAME

# Stream state from host for the movement assertion.
host STREAM_STATE interval_ms=50
host STATE
wait role=host contains="EVT name=STATE" timeout_ms=2000
sleep 100

# Log steerable actors once for debugging (controllers/owners).
host LIST_STEERABLE
sleep 200

# Sanity: simulation should actually advance on the host.
wait_state_change role=host field=tick min_abs_delta=1 timeout_ms=5000

# Force a sustained input sample for a short period to ensure the marble gets moving.
host OVERRIDE_INPUT player=0 ticks=120 fx=300.0 fy=0.0
sleep 50
host STATE
wait role=host contains="lc0_fx=300.000" timeout_ms=2000

# The host should observe its own p0_x changing.
wait_state_change role=host field=p0_x min_abs_delta=0.05 timeout_ms=5000

both QUIT
