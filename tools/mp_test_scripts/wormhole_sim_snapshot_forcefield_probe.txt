# Probe WormHole sim snapshot restore for force-field re-registration.
# Save while the wormhole is on, turn it off so the force field is removed, then
# load and confirm the restored on-state pulls and teleports the marble again.

both SET_BOOL key=MultiplayerAutoDetectConnectivity value=0
both SET_BOOL key=MultiplayerDebugVisualPrediction value=0
both SET_BOOL key=MultiplayerDebugZeroFillInputs value=1
both SET_BOOL key=MultiplayerDebugRollbackEnabled value=0
both SET_BOOL key=MultiplayerDebugNetSimEnabled value=0
both SET_BOOL key=MultiplayerDebugLogging value=1
both SET_INT key=MultiplayerDebugTickLengthMs value=10
both SET_INT key=MultiplayerDebugInputDelayTicks value=0

host START_HOST session=1 seed=1 expected=2 port=23485 host_id=mptest-host
client START_CLIENT session=1 seed=1 expected=2 port=23485 host_ip=127.0.0.1 host_id=mptest-host
wait role=client contains="EVT name=JOINED" timeout_ms=15000

host LOAD_LEVEL level_id="enigma_experimental/mptest_wormhole_snapshot_1"
sleep 1500
both START_GAME
host STREAM_STATE interval_ms=50
wait role=host contains="sv_world_init=1" timeout_ms=5000

host GET_CELL x=4 y=2
host SET_ACTOR_POS player=0 x=4.250 y=2.500 vx=0.0 vy=0.0
sleep 500
host GET_ACTOR_GRID player=0

host SET_ACTOR_POS player=0 x=1.500 y=3.500 vx=0.0 vy=0.0
host SIM_SNAPSHOT_SAVE

host SEND_CELL_MESSAGE x=4 y=2 layer=it message=close
host GET_CELL x=4 y=2

host SIM_SNAPSHOT_LOAD
host GET_CELL x=4 y=2
host SET_ACTOR_POS player=0 x=4.250 y=2.500 vx=0.0 vy=0.0
sleep 500
host GET_ACTOR_GRID player=0

both QUIT
